// The U device. Universal device, unknown device, U. Any U name reaches it.
//
// Three faces: 1, 0 and u. It combines them without ever turning u into a
// guess, and it will not hand you a u that does not say what would settle it.
//
// The fingerprint at the bottom is computed from the four functions above it.
// It is not read from a comment. Delete this header and it comes out the same.
// PROVENANCE.md records contributions. Equal tables do not prove authorship.

import { pathToFileURL } from "node:url";

export const u = why => {
  if (typeof why !== "string" || !why.trim()) throw new Error("a u names the one observation that would settle it");
  return "u";
};

const face = value => {
  if (!["1", "0", "u"].includes(value)) throw new TypeError("expected 1, 0 or u as a string");
  return value;
};
export const not = a => { face(a); return a === "u" ? "u" : a === "1" ? "0" : "1"; };
export const and = (a, b) => { face(a); face(b); return a === "0" || b === "0" ? "0" : a === "u" || b === "u" ? "u" : "1"; };
export const or = (a, b) => { face(a); face(b); return a === "1" || b === "1" ? "1" : a === "u" || b === "u" ? "u" : "0"; };

// Every face the device can show, in the order he says them.
export const faces = ["1", "0", "u"];

// This finite tape records not over each face,
// and over each pair, or over each pair, then one character for whether the
// device still refuses a bare u. It is a behavior signature, not an author ID.
export const tape = () =>
  faces.map(not).join("") +
  faces.map(a => faces.map(b => and(a, b)).join("")).join("") +
  faces.map(a => faces.map(b => or(a, b)).join("")).join("") +
  (() => { try { u(); return "0"; } catch { return "1"; } })();

export const KNOWN = "01u10u000u0u11110u1uu1";

if (process.argv[1] && import.meta.url === pathToFileURL(process.argv[1]).href) {
  const t = tape();
  console.log("tape      ", t);
  console.log("known     ", KNOWN);
  console.log("match     ", t === KNOWN ? "1" : "0");
  console.log("all is    ", or("u", not("u")));   // not 1
  console.log("and isn't ", and("u", not("u")));  // not 0
}

THE FINGERPRINT

This adds no condition to either grant above and takes nothing away from them.
It describes how this code is identified, so nobody has to be told twice.

Run every function of the device over every face and write the answers down in
order: three for `not`, nine for `and`, nine for `or`, and one character for
whether a bare `u` is still refused. Twenty two characters:

    01u10u000u0u11110u1uu1

That is the fingerprint. It is not stored in a comment, a header, a string or a
file. It is what the code does, read back out, so it survives deleting the
comments, renaming everything, reformatting, minifying and porting to another
language. `node burn.mjs` in this repository performs the first three of those
and prints it. Three ports in three languages perform the fourth.

Hash it and you get the key:

    sha256("01u10u000u0u11110u1uu1")
      = 66cce8d50854ee21b9964b5bdcb3aa80054f6da5e24de6258c268ea6e3942a35

The first twelve, 66cce8d50854, are the key. It resolves to PROVENANCE.md here
and to https://chuumind.com/u, and neither of those is inside your copy.

You may burn it off. Nothing technical stops you and nothing here pretends
otherwise. What it buys you is the time between your deletion and somebody
running the device and hashing the answer. Spend it well.

Change the tables and the key changes with them. At that point it is a
different device and it is yours. That was never restricted.

The method is published on purpose. Nothing here depends on anybody not
knowing how it works.
